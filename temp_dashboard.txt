<script>
function formatIsoToDisplay(isoStr) {
  if (!isoStr) return '';
  const dt = new Date(isoStr);
  const pad = (n) => n.toString().padStart(2, '0');
  return `${pad(dt.getDate())}/${pad(dt.getMonth()+1)}/${dt.getFullYear()} ${pad(dt.getHours())}:${pad(dt.getMinutes())}`;
}
window.addEventListener('load', () => {
  const stopModal = document.getElementById('stopModal');
  stopModal?.addEventListener('show.bs.modal', event => {
    const button = event.relatedTarget;
    const loomId = button.getAttribute('data-loom-id');
    const loomNumber = button.getAttribute('data-loom-number');
    const startTime = button.getAttribute('data-start-time');
    document.getElementById('stop_loom_id').value = loomId;
    document.getElementById('stop_loom_label').value = `Tear ${loomNumber}`;
    document.getElementById('stop_running_since').value = formatIsoToDisplay(startTime);
    document.getElementById('stopForm').action = `/looms/${loomId}/stop`;
    const reasonSelect = document.getElementById('stop_reason_select');
    const stopBtn = document.getElementById('stop_submit_btn');
    if (reasonSelect && stopBtn) {
      const toggle = () => { stopBtn.disabled = !reasonSelect.value; };
      reasonSelect.onchange = toggle;
      toggle();
    }
  });

  const startModal = document.getElementById('startModal');
  startModal?.addEventListener('show.bs.modal', event => {
    const button = event.relatedTarget;
    const loomId = button.getAttribute('data-loom-id');
    const loomNumber = button.getAttribute('data-loom-number');
    const stopTime = button.getAttribute('data-stop-time');
    const reason = button.getAttribute('data-reason') || '';
    const reasonId = button.getAttribute('data-reason-id') || '';
    document.getElementById('start_loom_id').value = loomId;
    document.getElementById('start_loom_label').value = `Tear ${loomNumber}`;
    document.getElementById('start_stopped_since').value = formatIsoToDisplay(stopTime);
    document.getElementById('start_reason_label').value = reason;
    document.getElementById('start_reason_id').value = reasonId;
    document.getElementById('startForm').action = `/looms/${loomId}/start`;
  });

  {% if open_modal %}
    const m = new bootstrap.Modal(document.getElementById('{{ open_modal }}Modal'));
    m.show();
  {% endif %}
});
</script>
